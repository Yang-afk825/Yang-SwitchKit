# ---- 全局开启环路检测 ----
# 锐捷的环路检测是全局开关，不能只对某几个端口开或关，开启后所有 up 的端口都会发检测报文。默认控制方式是「断开连接（shutdown）」，自动恢复时间默认 60 秒。
configure terminal
loopback-detection enable
loopback-detection interval 10
end
write

# ---- 边缘端口 + BPDU 保护 ----
# 边缘端口一旦收到 BPDU，说明下面接的不是终端而是交换机，设备会直接把该端口 err-disable。
configure terminal
interface range GigabitEthernet 1/0/1-22
 spanning-tree portfast
 spanning-tree bpduguard enable
 exit
end

# ---- BPDU 保护触发后的自动恢复 ----
# err-disable 的端口默认不会自己恢复，配了这两条才会定时重试。部分版本没有 errdisable 命令，下发前用 ? 确认。
configure terminal
errdisable recovery cause bpduguard
errdisable recovery interval 300
end

# ---- 核对结果 ----
# 查看全局环路检测状态与检测到的环路端口。
show loopback-detection
