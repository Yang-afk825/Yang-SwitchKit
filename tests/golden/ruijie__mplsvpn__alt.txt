# ---- 先建好 Loopback ----
# LSR ID 用 Loopback 地址最稳。锐捷的环回口写法是 Loopback 0，接口名和编号之间有空格。
configure terminal
interface Loopback 0
 ip address 1.1.1.1 255.255.255.255
 exit
end

# ---- 开启 MPLS 与 LDP ----
# 锐捷的写法和其他三家都不一样：全局是 mpls ip 开转发，mpls router ldp 进 LDP 视图再用 ldp router-id 指定 Router ID，没有 mpls lsr-id 这条命令。接口下要 mpls ip 加 label-switch 两条。
configure terminal
mpls ip
mpls router ldp
 ldp router-id interface Loopback 0 force
 exit
interface Vlan 100
 ip address 10.0.0.1 255.255.255.252
 mpls ip
 label-switch
 exit
end

# ---- 创建 VRF ----
# 锐捷用 ip vrf（与思科一致）。
configure terminal
ip vrf VPNA
 rd 100:1
 route-target export 100:1
 route-target import 100:1
 exit
end

# ---- 把 CE 接口划入 VRF ----
# 注意顺序：ip vrf forwarding 会清掉接口上原有的 IP 配置，所以必须先划 VRF 再配地址，反了地址就没了。
configure terminal
interface Vlan 10
 ip vrf forwarding VPNA
 ip address 10.1.1.2 255.255.255.0
 exit
end

# ---- MP-BGP 传递 VPN 路由 ----
# 锐捷的 VPNv4 地址族写成 address-family vpnv4 unicast，比思科多一个 unicast。
configure terminal
router bgp 100
 no bgp default ipv4-unicast
 neighbor 2.2.2.2 remote-as 100
 neighbor 2.2.2.2 update-source Loopback 0
 address-family vpnv4 unicast
  neighbor 2.2.2.2 activate
  neighbor 2.2.2.2 send-community extended
  exit-address-family
 address-family ipv4 vrf VPNA
  neighbor 10.1.1.1 remote-as 65410
  neighbor 10.1.1.1 activate
  redistribute connected
  exit-address-family
 exit
end

# ---- 保存配置 ----
end
write

# ---- 分段核对 ----
# 按顺序查：LDP 邻居通了没、标签有没有下发、VRF 建了没、MP-BGP 邻居起来没、VPN 路由收没收到。
show mpls ldp neighbor
show mpls ldp binding
show ip vrf
show bgp vpnv4 unicast summary
show ip route vrf VPNA

# [风险] MPLS 的前提是公网侧 IGP 已经跑通，而且两端的 Loopback 都要被宣告进 IGP，否则 LDP 会话建不起来。建议先配 OSPF 或 IS-IS 那两条场景。
