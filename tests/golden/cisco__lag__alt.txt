# ---- 成员端口加入通道组 ----
# LACP 用 mode active，静态聚合用 mode on；两端模式必须匹配。
configure terminal
interface range GigabitEthernet1/0/1 - 2
 channel-group 1 mode on
 exit

# ---- 配置 Port-channel 接口 ----
# 通道口的 VLAN 配置要与成员口一致。
configure terminal
interface Port-channel1
 exit

# ---- 保存配置 ----
end
write memory
