# ---- 启用 OSPF 并指定 Router ID ----
# Router ID 建议手工指定成 Loopback 地址，不要依赖设备自动选举。
system-view
ospf 1 router-id 1.1.1.1
 quit

# ---- 宣告网段 ----
# VRP 的 area 视图下 network 用反掩码（通配符掩码）。
system-view
ospf 1
 area 0
  network 192.168.10.0 0.0.0.255
  quit
 area 0
  network 192.168.20.0 0.0.0.255
  quit
 quit

# ---- 设置被动接口 ----
# 被动接口不发 Hello，但网段照样宣告出去，适合接终端的接口。
system-view
ospf 1
 silent-interface Vlan-interface 10
 quit

# ---- 保存配置 ----
return
save

# ---- 核对邻居与路由 ----
# 先看邻居是不是 Full，再看路由有没有进表。
system-view
display ospf peer
display ospf routing
