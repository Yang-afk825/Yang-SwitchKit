# ---- 配置 VRRP 虚拟网关 ----
# 两台设备的 vrid、虚拟 IP 必须完全一致，否则会形成双主。
system-view
interface Vlan-interface 10
 ip address 192.168.10.1 24
 vrrp vrid 10 virtual-ip 192.168.10.254
 vrrp vrid 10 priority 120
 vrrp vrid 10 preempt-mode delay 5
 quit

# ---- 保存配置 ----
return
save force
