# ---- 创建 VLAN ----
# 华为的 VLAN 描述字段是 description，H3C 与思科用 name，注意区别。
system-view
vlan 10
 quit
vlan 20
 quit
vlan 30
 quit

# ---- 配置 Trunk 端口 ----
interface range GigabitEthernet1/0/1 to GigabitEthernet1/0/8
 port link-type trunk
 port trunk allow-pass vlan all
 quit

# ---- 保存配置 ----
# save 命令会提示确认，按 Y 继续。
return
save

# [风险] Trunk 放通全部 VLAN 会扩大广播域，建议改成明确的 VLAN 列表。
