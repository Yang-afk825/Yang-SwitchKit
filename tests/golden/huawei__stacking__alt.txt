# ---- 设置堆叠优先级 ----
# slot 0 为本机槽位号；华为 iStack 的槽位号需先用 display stack 确认。
system-view
stack slot 0 priority 32

# ---- 绑定堆叠物理端口 ----
# 华为堆叠端口编号与物理端口需一一对应，请按机型手册核对。
system-view
interface stack-port 0/1
 port interface XGigabitEthernet1/0/49 enable
 quit
interface stack-port 0/2
 port interface XGigabitEthernet1/0/50 enable
 quit

# ---- 保存并重启 ----
# 保存后执行 reboot 使堆叠配置生效。
system-view
save
