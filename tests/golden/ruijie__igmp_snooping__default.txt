# ---- 开启 IGMP Snooping ----
# 全局一条打开，再逐 VLAN 打开。
configure terminal
ip igmp snooping
ip igmp snooping vlan 10
end

# ---- 开启 IGMP 查询器 ----
# VLAN 内没有三层组播设备时，由交换机自己发查询报文维持组播表项。
configure terminal
ip igmp snooping querier
ip igmp snooping vlan 10 querier
end

# ---- 静态指定路由器端口 ----
# 组播流量从这些口进来。不指定的话靠动态学习，上行不稳时表项容易丢。
configure terminal
ip igmp snooping vlan 10 mrouter interface GigabitEthernet 1/0/24
end

# ---- 保存配置 ----
end
write
