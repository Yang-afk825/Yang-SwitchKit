# ---- 创建 VLAN ----
configure terminal
vlan 10
exit
vlan 20
exit
vlan 30
exit

# ---- Access 端口划入 VLAN 10 ----
interface range GigabitEthernet 1/0/1-8
 switchport mode access
 switchport access vlan 10
 exit

# ---- 保存配置 ----
end
write
