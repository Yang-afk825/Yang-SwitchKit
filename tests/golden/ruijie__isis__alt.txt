# ---- 启用 IS-IS 并配置 NET ----
# 锐捷与思科同源，层级关键字是 level-2-only，度量风格是 metric-style。部分 RGOS 版本的 metric-style 命令名不同，下发前用 ? 确认。
configure terminal
ip routing
router isis
 net 49.0001.0000.0000.0001.00
 is-type level-2-only
 metric-style narrow
 exit
end

# ---- 配置验证钥匙串 ----
# 锐捷的 IS-IS 验证走钥匙串，与思科一致。
configure terminal
key chain ISIS-AUTH
 key 1
  key-string Isis@2026
  exit
 exit
end

# ---- 三层接口配置 ----
# 接口下用 ip router isis 使能。
configure terminal
interface Vlan 10
 ip router isis
 isis circuit-type level-2-only
 isis authentication mode md5
 isis authentication key-chain ISIS-AUTH
 exit
interface Vlan 20
 ip router isis
 isis circuit-type level-2-only
 isis authentication mode md5
 isis authentication key-chain ISIS-AUTH
 exit
end

# ---- 保存配置 ----
end
write

# ---- 核对邻居与路由 ----
# 先看邻接是不是 Up，再看路由表里有没有学到对端网段。
show isis neighbors
show ip route isis
