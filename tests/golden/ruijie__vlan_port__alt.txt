# ====================================================================
# 锐捷 Ruijie | VLAN 创建与端口划分
# 由 Yang-SwitchKit 生成，执行前请核对版本与端口名
# ====================================================================

# ---- 创建 VLAN ----
configure terminal
vlan 10
exit
vlan 20
exit
vlan 30
exit

# ---- 配置 Trunk 端口 ----
# 锐捷追加放通 VLAN 必须带 add，否则会覆盖已有列表。
interface range GigabitEthernet 1/0/1-8
 switchport mode trunk
 switchport trunk allowed vlan all
 exit

# ---- 保存配置 ----
end
write

# [风险] Trunk 放通全部 VLAN 会扩大广播域，建议改为明确的 VLAN 列表。
