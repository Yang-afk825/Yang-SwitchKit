# ---- 配置网关冗余 ----
# HSRP 是思科私有协议，与 VRRP 不能互通。默认优先级 100，范围 0-255。
configure terminal
interface Vlan10
 ip address 192.168.10.1 255.255.255.0
 no shutdown
 standby 10 ip 192.168.10.254
 standby 10 priority 120
 exit

# ---- 保存配置 ----
end
write memory
