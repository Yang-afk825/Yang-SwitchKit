# ---- 配置 RADIUS 服务器模板 ----
# 华为要先建 RADIUS 服务器模板，再让认证域去引用它。
system-view
radius-server template SCG-RADIUS
 radius-server authentication 192.168.1.30 1812
 radius-server accounting 192.168.1.30 1813
 radius-server shared-key cipher Radius@123
 quit

# ---- 配置认证方案与认证域 ----
# 认证方案决定用哪种方式认证，认证域把方案和 RADIUS 模板绑在一起。终端上填的账号要带上 @域名 后缀，或者把该域设为默认域。
system-view
aaa
 authentication-scheme SCG-AUTH
  authentication-mode radius
  quit
 domain scg.local
  authentication-scheme SCG-AUTH
  radius-server SCG-RADIUS
  quit
 quit

# ---- 端口开启 802.1X ----
# 华为是接口下 dot1x enable，没有全局开关。
system-view
interface range GigabitEthernet1/0/1 to GigabitEthernet1/0/8
 dot1x enable
 dot1x port-method macbased
 dot1x port-control auto
 quit

# ---- 保存配置 ----
return
save

# ---- 核对认证状态 ----
# 先看端口有没有在跑 802.1X，再看在线用户列表。
system-view
display dot1x
display access-user
