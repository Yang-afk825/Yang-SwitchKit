# ---- 配置网关冗余 ----
# VRRP 可与华为 / H3C / 锐捷设备互通，跨厂商组网请统一使用 VRRP。
configure terminal
interface Vlan10
 ip address 192.168.10.1 255.255.255.0
 no shutdown
 vrrp 10 ip 192.168.10.254
 vrrp 10 priority 120
 vrrp 10 preempt
 exit

# ---- 保存配置 ----
end
write memory

# [提示] 主备优先级要拉开差距，避免频繁切换。
# [提示] 只有思科 / 锐捷支持 HSRP；VRRP 四家通用。
