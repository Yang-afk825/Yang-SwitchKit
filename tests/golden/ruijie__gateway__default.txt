# ---- 配置网关冗余 ----
# 锐捷的 VRRP 语法与思科一致，能与其他厂商的 VRRP 互通。
configure terminal
interface Vlan 10
 ip address 192.168.10.1 255.255.255.0
 vrrp 10 ip 192.168.10.254
 vrrp 10 priority 120
 vrrp 10 preempt
 exit

# ---- 保存配置 ----
end
write
