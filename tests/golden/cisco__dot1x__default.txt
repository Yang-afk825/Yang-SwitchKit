# ---- 配置 RADIUS 服务器 ----
# 思科要先 aaa new-model 打开 AAA，再定义 radius server，最后用 dot1x system-auth-control 全局开启 802.1X。老版本 IOS 用的是 radius-server host ... key ... 那套命令。
configure terminal
aaa new-model
radius server SCG-RADIUS
 address ipv4 192.168.1.30 auth-port 1812 acct-port 1813
 key Radius@123
 exit
aaa authentication dot1x default group radius
aaa authorization network default group radius
dot1x system-auth-control
end

# ---- 端口开启 802.1X ----
# dot1x pae authenticator 表示这个端口扮演认证方角色，思科的 802.1X 必须配它。port-control auto 才是「认证通过才放行」。
configure terminal
interface range GigabitEthernet1/0/1 - 8
 switchport mode access
 authentication port-control auto
 dot1x pae authenticator
 dot1x host-mode multi-host
 exit
end

# ---- 保存配置 ----
end
write memory

# ---- 核对认证状态 ----
# 看端口状态是 AUTHORIZED 还是 UNAUTHORIZED。
show dot1x all summary
show authentication sessions
