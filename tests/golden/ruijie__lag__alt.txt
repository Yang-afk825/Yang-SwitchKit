# ---- 成员端口加入聚合组 ----
# 锐捷的聚合口叫 AggregatePort（AP），成员口用 port-group 加入。LACP 用 mode active，静态聚合用 mode on。
configure terminal
interface range GigabitEthernet 1/0/1-2
 port-group 1 mode on
 exit

# ---- 配置聚合口 ----
# 聚合口的 VLAN 配置必须与成员口一致，否则无法正常转发。
configure terminal
interface AggregatePort 1
 exit

# ---- 保存配置 ----
end
write

# [风险] 若设备不支持 port-group mode 参数，请去掉 mode 部分直接写 port-group 组号。

# --------------------------------------------------------------------

# [提示] 跨设备聚合需要堆叠或 M-LAG / VSU 支持，单机聚合不跨设备。
