# ====================================================================

# ---- 创建 VLAN ----
# 华为的 VLAN 描述字段是 description，H3C 与思科用 name，注意区别。
system-view
vlan 10
 quit
vlan 20
 quit
vlan 30
 quit

# ---- Access 端口划入 VLAN 10 ----
# 华为的 Access 端口用 port default vlan，H3C 是 port access vlan。
interface range GigabitEthernet1/0/1 to GigabitEthernet1/0/8
 port link-type access
 port default vlan 10
 quit

# ---- 保存配置 ----
# save 命令会提示确认，按 Y 继续。
return
save

# [风险] VLAN 有 3 个、端口有 8 个，无法一一对应，已统一划入 VLAN 10。
