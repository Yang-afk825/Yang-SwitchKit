# ---- 启用 IS-IS 并配置 NET ----
# 官方建议 cost-style、is-level、network-entity 三条一起改时把 network-entity 放最后，避免配置顺序不对导致 IS-IS 进程重启。
system-view
isis 1
 is-level level-2
 cost-style wide
 network-entity 49.0001.0000.0000.0001.00
 quit

# ---- 接口使能 IS-IS ----
# 只能建单层邻接时用 isis circuit-level 限制，能少收一半的 PDU。
system-view
interface Vlan-interface 10
 isis enable 1
 quit
interface Vlan-interface 20
 isis enable 1
 quit

# ---- 引入直连路由 ----
# import-route 必须敲在 address-family ipv4 视图里，在 isis 主视图敲会报 Unrecognized command。
system-view
isis 1
 address-family ipv4
  import-route direct
  quit
 quit

# ---- 保存配置 ----
return
save force

# ---- 核对邻居与路由 ----
# 先看邻接是不是 Up，再看路由表里有没有学到对端网段。
system-view
display isis peer
display isis route
display isis lsdb
