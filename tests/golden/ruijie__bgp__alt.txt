# ---- 配置 BGP 与邻居 ----
# 锐捷的 BGP 语法与思科基本一致，邻居命令是 neighbor。三层交换机别忘了 ip routing。
configure terminal
ip routing
router bgp 65001
 bgp router-id 1.1.1.1
 bgp log-neighbor-changes
 neighbor 10.0.0.2 remote-as 65002
 neighbor 10.0.0.2 password Bgp@2026
 neighbor 10.0.0.3 remote-as 65003
 neighbor 10.0.0.3 password Bgp@2026
 network 192.168.10.0 mask 255.255.255.0
 redistribute connected
 exit
end

# ---- 保存配置 ----
end
write

# ---- 核对邻居与路由 ----
# State/PfxRcd 那一列出现数字就是 Established。
show bgp summary
show bgp ipv4 unicast
