# ---- 启用 OSPF 并宣告网段 ----
# 锐捷的 OSPF 语法与思科基本一致，network 一条带上区域号，用反掩码。三层交换机别忘了 ip routing。
configure terminal
ip routing
router ospf 1
 router-id 1.1.1.1
 network 192.168.10.0 0.0.0.255 area 0
 network 192.168.20.0 0.0.0.255 area 0
 area 0 authentication message-digest
 end
write

# ---- 设置被动接口 ----
# 被动接口不发 Hello，但网段照样宣告出去，适合接终端的接口。
configure terminal
router ospf 1
 passive-interface Vlan-interface 10
end

# ---- 接口上配置 MD5 密钥 ----
# 这一段是注释，不会下发；请按实际三层接口补配。
! 区域开了 MD5 验证后，还要在每个跑 OSPF 的接口上配密钥，
! 命令形如（在 VLAN 接口视图下）：
!   ip ospf authentication message-digest
!   ip ospf message-digest-key 1 md5 Ospf@2026
! 密钥必须两端一致。本工具不知道你的三层接口清单，所以只给模板。

# ---- 核对邻居与路由 ----
# 先看邻居是不是 FULL，再看 OSPF 路由有没有进表。
show ip ospf neighbor
show ip route ospf
