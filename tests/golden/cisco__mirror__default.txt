# ---- 配置镜像会话 ----
# 思科的镜像方向用 rx / tx / both，与其他厂商的 inbound / outbound 写法不同。
configure terminal
monitor session 1 source interface GigabitEthernet1/0/1 - 4 both
monitor session 1 destination interface GigabitEthernet1/0/24
 exit

# ---- 保存配置 ----
end
