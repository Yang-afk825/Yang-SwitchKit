# ---- 开启动态 ARP 检测（DAI） ----
# DAI 靠 DHCP Snooping 绑定表判断报文是否合法，所以要先开 DHCP Snooping 并指定信任端口。validate 三个关键字分别校验源 MAC、目的 MAC、IP 是否与 ARP 内容一致。
configure terminal
ip dhcp snooping
ip dhcp snooping vlan 10,20
no ip dhcp snooping information option
ip arp inspection vlan 10,20
ip arp inspection validate src-mac dst-mac ip
end

# ---- 指定 ARP 信任端口 ----
# 上行口必须同时设为 DHCP Snooping 和 DAI 的信任口，否则来自网关的正常 ARP 会被丢弃。
configure terminal
interface GigabitEthernet1/0/24
 ip dhcp snooping trust
 ip arp inspection trust
 exit
end

# ---- 用户口开启 IP 源防护 ----
# 按绑定表校验源 IP（和源 MAC），终端私自改 IP 后报文会被直接丢弃。
configure terminal
interface range GigabitEthernet1/0/1 - 22
 ip verify source
 ip verify source port-security
 exit
end

# ---- 保存配置 ----
end
write memory
