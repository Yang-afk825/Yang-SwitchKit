# ---- 开启动态 ARP 检测（DAI） ----
# DAI 靠 DHCP Snooping 绑定表判断报文是否合法，所以要先开 DHCP Snooping 并指定信任端口。
configure terminal
ip dhcp snooping
ip dhcp snooping vlan 10,20
ip arp inspection vlan 10,20
end

# ---- 指定信任端口 ----
# 上行口必须同时设为两者的信任口，否则来自网关的正常 ARP 会被丢弃。
configure terminal
interface GigabitEthernet 1/0/24
 ip dhcp snooping trust
 ip arp inspection trust
 exit
end

# ---- 保存配置 ----
end
write

# ---- 核对结果 ----
# 确认生效的 VLAN 列表与信任端口是否正确。
show ip arp inspection
