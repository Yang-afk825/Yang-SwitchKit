# ---- 配置本地日志 ----
# service timestamps ... localtime 让日志带本地时间，没有时间戳的日志在排障时几乎用不上。思科默认不往 console 打日志，要开就加 logging console。
configure terminal
logging buffered 512
clock timezone UTC 8
logging console
end

# ---- 配置日志服务器 ----
# 只把 error 及以上级别的日志送到服务器，避免刷屏。logging source-interface 请改成实际的管理口。
configure terminal
logging host 10.0.0.7
logging trap error
logging source-interface Vlan1
end

# ---- NTP 时间同步 ----
# 时钟不准，日志的先后顺序就不可信，NTP 要和日志一起配。
configure terminal
ntp server 10.0.0.5
end

# ---- 保存配置 ----
end
write memory

# ---- 验证日志是否送达 ----
# 配完在设备上制造一条日志（比如插拔一个端口），去服务器上确认收到了。
show logging
show clock
