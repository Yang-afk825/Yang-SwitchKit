# ---- 定义 ACL ----
# 思科的扩展 ACL 编号为 100-199 或 2000-2699；命名的写法则用名称代替编号。
configure terminal
ip access-list extended 100
 10 permit ip 192.168.10.0 0.0.0.255 any
 20 permit ip 192.168.20.0 0.0.0.255 192.168.30.0 0.0.0.255
 30 deny ip any any
 exit

# ---- 应用到端口 ----
# 思科用 ip access-group 调用 ACL，H3C 用 packet-filter，华为用 traffic-filter。
configure terminal
interface GigabitEthernet1/0/24
 ip access-group 100 in
 exit

# ---- 保存配置 ----
end
write memory
