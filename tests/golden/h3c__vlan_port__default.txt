# ---- 创建 VLAN ----
system-view
vlan 10
 quit
vlan 20
 quit
vlan 30
 quit

# ---- Access 端口划入 VLAN 10 ----
interface range GigabitEthernet1/0/1 to GigabitEthernet1/0/8
 port link-type access
 port access vlan 10
 quit

# ---- 保存配置 ----
return
save force
