# ---- 开启 QoS 并配置信任模式 ----
# 锐捷与思科同源，先 mls qos 开总开关，再配信任模式。802.1p 对应的关键字是 cos。
configure terminal
mls qos
interface range GigabitEthernet 1/0/1-24
 mls qos trust dscp
 exit
end

# ---- 保存配置 ----
end
write

# ---- 核对结果 ----
# 确认信任模式生效、策略有命中计数。
show mls qos interface
show policy-map interface
