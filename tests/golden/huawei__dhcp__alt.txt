# ---- 配置 DHCP 中继 ----
# 华为中继用 dhcp relay server-ip，H3C 用 dhcp relay server-address。
system-view
dhcp enable
interface Vlanif 10
 ip address 192.168.10.1 24
 dhcp select relay
 dhcp relay server-ip 10.0.0.100
 quit

# ---- 保存配置 ----
# save 命令会提示确认，按 Y 继续。
return
save
