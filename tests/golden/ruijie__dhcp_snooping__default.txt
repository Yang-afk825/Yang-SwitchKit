# ---- 开启 DHCP Snooping ----
# ip dhcp snooping 是总开关，再按 VLAN 打开。默认会校验请求报文里的客户端 MAC 与源 MAC 是否一致，本工具按勾选情况生成。
configure terminal
ip dhcp snooping
ip dhcp snooping vlan 10,20
end

# ---- 指定信任端口 ----
# 只有信任端口上收到的 DHCP 应答才会被转发，其余一律丢弃。
configure terminal
interface GigabitEthernet 1/0/24
 ip dhcp snooping trust
 exit
end

# ---- 保存配置 ----
end
write

# ---- 查看绑定表 ----
# 绑定表是动态 ARP 检测和 IP 源防护的判断依据。
show ip dhcp snooping binding
