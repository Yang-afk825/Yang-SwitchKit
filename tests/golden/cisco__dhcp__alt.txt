# ---- 配置 DHCP 中继 ----
# 思科用 ip helper-address 转发 DHCP 请求，H3C/华为用专用的 relay 命令。
configure terminal
interface Vlan10
 ip address 192.168.10.1 255.255.255.0
 ip helper-address 10.0.0.100
 exit

# ---- 保存配置 ----
end
write memory

# [提示] 中继模式只需在三层接口上指定服务器地址，不需要建地址池。
