# ---- 配置 DHCP 服务器 ----
# 华为的三层接口名是 Vlanif，H3C 是 Vlan-interface，思科是 Vlan。
system-view
dhcp enable
ip pool POOL-VLAN10
 network 192.168.10.0 mask 255.255.255.0
 gateway-list 192.168.10.1
 dns-list 223.5.5.5
 excluded-ip-address 192.168.10.1 192.168.10.20
 quit
interface Vlanif 10
 ip address 192.168.10.1 24
 dhcp select global
 quit

# ---- 保存配置 ----
# save 命令会提示确认，按 Y 继续。
return
save
