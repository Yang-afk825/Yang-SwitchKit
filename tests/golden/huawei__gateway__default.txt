# ---- 配置 VRRP 虚拟网关 ----
# 华为的端口联动直接写在 vrrp 命令里，不需要先建 track 项。
system-view
interface Vlanif 10
 ip address 192.168.10.1 24
 vrrp vrid 10 virtual-ip 192.168.10.254
 vrrp vrid 10 priority 120
 vrrp vrid 10 preempt-mode timer delay 5
 quit

# ---- 保存配置 ----
# save 命令会提示确认，按 Y 继续。
return
save
