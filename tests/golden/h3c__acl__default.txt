# ---- 定义 ACL ----
# Comware V5 使用 acl number 编号，V7 使用 acl basic / advanced。
system-view
acl advanced 3000
 step 10
 rule 10 permit ip source 192.168.10.0 0.0.0.255
 rule 20 permit ip source 192.168.20.0 0.0.0.255 destination 192.168.30.0 0.0.0.255
 rule 30 deny ip
 quit

# ---- 应用到端口 ----
system-view
interface GigabitEthernet1/0/24
 packet-filter 3000 inbound
 quit

# ---- 保存配置 ----
return
save force
